// ── Entry points ───────────────────────────────────────────── function doGet(e) { const action = e.parameter.action; try { if (action === 'getItems') return jsonResponse(getItems()); if (action === 'getLog') return jsonResponse(getLog()); return jsonResponse({ error: 'Unknown action' }); } catch (err) { return jsonResponse({ error: err.message }); } } function doPost(e) { const data = JSON.parse(e.postData.contents); const action = data.action; try { if (action === 'checkout') return jsonResponse(logCheckout(data)); if (action === 'returnItem') return jsonResponse(logReturn(data)); if (action === 'updateStock') return jsonResponse(updateStock(data)); return jsonResponse({ error: 'Unknown action' }); } catch (err) { return jsonResponse({ error: err.message }); } } // ── Sheet helpers ───────────────────────────────────────────── function getOrCreateSheet(name, headers) { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length) .setFontWeight('bold') .setBackground('#f3f3f3'); } return sheet; } function seedInventoryIfEmpty(sheet) { if (sheet.getLastRow() <= 1) { DEFAULT_ITEMS.forEach(item => { sheet.appendRow([item.id, item.cat, item.name, item.qty]); }); } } // ── Actions ─────────────────────────────────────────────────── function getItems() { const sheet = getOrCreateSheet(INVENTORY_SHEET, ['ID', 'Category', 'Name', 'Qty']); seedInventoryIfEmpty(sheet); const rows = sheet.getDataRange().getValues(); const items = rows.slice(1).map(r => ({ id: r[0], cat: r[1], name: r[2], qty: r[3], })); return { items }; } function getLog() { const sheet = getOrCreateSheet(LOG_SHEET, ['Timestamp', 'Item', 'Who', 'Qty', 'Date', 'Note']); if (sheet.getLastRow() <= 1) return { log: [] }; const rows = sheet.getDataRange().getValues(); const log = rows.slice(1).reverse().map(r => ({ ts: r[0], itemName: r[1], who: r[2], qty: r[3], date: r[4], note: r[5], })); return { log }; } function logCheckout(data) { // Deduct from inventory const invSheet = getOrCreateSheet(INVENTORY_SHEET, ['ID', 'Category', 'Name', 'Qty']); const rows = invSheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] == data.itemId) { const newQty = Math.max(0, rows[i][3] - data.qty); invSheet.getRange(i + 1, 4).setValue(newQty); break; } } // Append to log const logSheet = getOrCreateSheet(LOG_SHEET, ['Timestamp', 'Type', 'Item', 'Who', 'Qty', 'Date', 'Note']); logSheet.appendRow([ new Date().toISOString(), 'CHECKOUT', data.itemName, data.who, data.qty, data.date, data.note || '', ]); return { success: true }; } function logReturn(data) { // Add back to inventory const invSheet = getOrCreateSheet(INVENTORY_SHEET, ['ID', 'Category', 'Name', 'Qty']); const rows = invSheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] == data.itemId) { const newQty = rows[i][3] + data.qty; invSheet.getRange(i + 1, 4).setValue(newQty); break; } } // Append to log const logSheet = getOrCreateSheet(LOG_SHEET, ['Timestamp', 'Type', 'Item', 'Who', 'Qty', 'Date', 'Note']); logSheet.appendRow([ new Date().toISOString(), 'RETURN', data.itemName, data.who, data.qty, data.date, data.note || '', ]); return { success: true }; } const sheet = getOrCreateSheet(INVENTORY_SHEET, ['ID', 'Category', 'Name', 'Qty']); const rows = sheet.getDataRange().getValues(); data.updates.forEach(update => { for (let i = 1; i < rows.length; i++) { if (rows[i][0] == update.id) { sheet.getRange(i + 1, 4).setValue(update.qty); break; } } }); return { success: true }; } function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON);
